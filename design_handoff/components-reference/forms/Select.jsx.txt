import React from 'react';

/** Labelled <select> with the brand's custom gold chevron and navy field styling. */
export function Select({ label, id, required = false, hint, children, className = '', ...rest }) {
  const inputId = id || (label ? `fx-${label.replace(/\s+/g, '-').toLowerCase()}` : undefined);
  return (
    <div className={className}>
      {label && (
        <label className="fx-label" htmlFor={inputId}>
          {label} {required && <span className="fx-label__req">*</span>}
        </label>
      )}
      <select id={inputId} className="fx-field" required={required} {...rest}>
        {children}
      </select>
      {hint && <p style={{ marginTop: 'var(--space-2)', fontSize: 'var(--text-xs)', color: 'var(--text-muted)' }}>{hint}</p>}
    </div>
  );
}
