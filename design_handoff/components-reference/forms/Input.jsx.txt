import React from 'react';

/**
 * Labelled text/number input field. Wrap a label + <input> with the brand's
 * navy field styling and gold focus ring.
 */
export function Input({
  label,
  id,
  required = false,
  hint,
  className = '',
  ...rest
}) {
  const inputId = id || (label ? `fx-${label.replace(/\s+/g, '-').toLowerCase()}` : undefined);
  return (
    <div className={className}>
      {label && (
        <label className="fx-label" htmlFor={inputId}>
          {label} {required && <span className="fx-label__req">*</span>}
        </label>
      )}
      <input id={inputId} className="fx-field" required={required} {...rest} />
      {hint && <p style={{ marginTop: 'var(--space-2)', fontSize: 'var(--text-xs)', color: 'var(--text-muted)' }}>{hint}</p>}
    </div>
  );
}
