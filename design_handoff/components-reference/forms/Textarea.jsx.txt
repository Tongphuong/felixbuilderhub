import React from 'react';

/** Labelled multi-line text area, sharing the navy field styling. */
export function Textarea({ label, id, required = false, hint, className = '', rows = 3, ...rest }) {
  const inputId = id || (label ? `fx-${label.replace(/\s+/g, '-').toLowerCase()}` : undefined);
  return (
    <div className={className}>
      {label && (
        <label className="fx-label" htmlFor={inputId}>
          {label} {required && <span className="fx-label__req">*</span>}
        </label>
      )}
      <textarea id={inputId} className="fx-field" rows={rows} required={required} {...rest} />
      {hint && <p style={{ marginTop: 'var(--space-2)', fontSize: 'var(--text-xs)', color: 'var(--text-muted)' }}>{hint}</p>}
    </div>
  );
}
