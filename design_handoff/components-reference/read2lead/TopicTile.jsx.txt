import React from 'react';

/** Selectable topic tile (emoji + label) used in the Read2Lead lesson builder. */
export function TopicTile({ emoji, label, selected = false, as = 'button', className = '', ...rest }) {
  const Tag = as;
  return (
    <Tag
      type={as === 'button' ? 'button' : undefined}
      className={['fx-topic', selected ? 'fx-topic--selected' : '', className].filter(Boolean).join(' ')}
      aria-pressed={as === 'button' ? selected : undefined}
      {...rest}
    >
      <span className="fx-topic__emoji" aria-hidden="true">{emoji}</span>
      <span className="fx-topic__label">{label}</span>
    </Tag>
  );
}
