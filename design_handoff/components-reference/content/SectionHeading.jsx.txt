import React from 'react';

/** Eyebrow + title + lead block. The standard way the brand opens a section. */
export function SectionHeading({ eyebrow, title, lead, align = 'left', titleSize = 'var(--text-display-sm)', className = '', ...rest }) {
  return (
    <div className={className} style={{ textAlign: align, maxWidth: align === 'center' ? '42rem' : undefined, marginInline: align === 'center' ? 'auto' : undefined }} {...rest}>
      {eyebrow && <p className="fx-eyebrow">{eyebrow}</p>}
      <h2 className="fx-section-heading__title" style={{ fontSize: titleSize }}>{title}</h2>
      {lead && <p className="fx-section-heading__lead">{lead}</p>}
    </div>
  );
}
