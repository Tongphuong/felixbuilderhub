/* Read2Lead UI kit — the AI lesson builder flow.
   Recreates the production /read2lead create-lesson experience:
   enter student code → choose a topic → generate → lesson ready.
   Plus a compact gamified app header and a leaderboard peek. */

(() => {
const { Button, Input, Card, Badge, RankBadge, ProgressBar, TopicTile } = window.FelixCoachingDesignSystem_05c199;

const TOPICS = [
  ['🐾', 'Động vật & thú cưng'], ['👨‍👩‍👧', 'Gia đình & bạn bè'], ['🏫', 'Trường học'],
  ['⚽', 'Thể thao'], ['🎮', 'Trò chơi & đồ chơi'], ['🚗', 'Xe cộ'],
  ['🍳', 'Đồ ăn & nấu ăn'], ['🌳', 'Thiên nhiên'], ['🎨', 'Vẽ & sáng tạo'],
  ['🦸', 'Nghề nghiệp & ước mơ'], ['✨', 'Tưởng tượng'], ['🏮', 'Lễ hội Việt Nam'],
];

function AppHeader({ onLeaderboard }) {
  return (
    <header style={{ position: 'sticky', top: 0, zIndex: 50, borderBottom: '1px solid var(--border-subtle)', background: 'color-mix(in srgb, var(--navy-950) 95%, transparent)', backdropFilter: 'blur(8px)' }}>
      <div style={{ maxWidth: 'var(--container)', margin: '0 auto', padding: '10px var(--gutter)', display: 'flex', alignItems: 'center', gap: 12 }}>
        <img src="../../assets/logo-header.png" alt="Read2Lead" style={{ height: 30, borderRadius: 5 }} />
        <div style={{ flex: 1 }} />
        <RankBadge tier="gold" level={7} />
        <span className="r2l-stat" style={{ fontWeight: 700, color: 'var(--gold)', fontSize: 'var(--text-sm)' }}>240 xu</span>
        <span className="r2l-stat" style={{ fontWeight: 700, color: 'var(--cream)', fontSize: 'var(--text-sm)' }}>🔥 5 ngày</span>
        <Button size="sm" variant="secondary" onClick={onLeaderboard}>Bảng xếp hạng</Button>
      </div>
      <div style={{ maxWidth: 'var(--container)', margin: '0 auto', padding: '0 var(--gutter) 10px' }}>
        <div style={{ display: 'flex', justifyContent: 'space-between', fontSize: 'var(--text-xs)', color: 'var(--cream-muted)', marginBottom: 4 }}>
          <span>Cấp 7 · Vàng</span><span>42/60 XP</span>
        </div>
        <ProgressBar value={70} />
      </div>
    </header>
  );
}

function Stepper({ step }) {
  const steps = ['Nhập mã', 'Chọn chủ đề', 'Tạo bài', 'Mở bài'];
  return (
    <div style={{ display: 'flex', alignItems: 'flex-start', justifyContent: 'space-between', gap: 4, marginBottom: 28 }}>
      {steps.map((label, i) => (
        <React.Fragment key={i}>
          <div style={{ flex: 1, textAlign: 'center' }}>
            <span style={{ display: 'flex', width: 34, height: 34, margin: '0 auto', alignItems: 'center', justifyContent: 'center', borderRadius: '50%', fontWeight: 700,
              background: i <= step ? 'var(--accent)' : 'color-mix(in srgb, var(--cream) 12%, transparent)',
              color: i <= step ? 'var(--navy-950)' : 'var(--cream)' }}>{i + 1}</span>
            <p style={{ marginTop: 6, fontSize: 'var(--text-xs)', color: i <= step ? 'var(--cream)' : 'var(--cream-dim)' }}>{label}</p>
          </div>
          {i < steps.length - 1 && <div style={{ height: 1, flex: 1, marginTop: 17, background: 'color-mix(in srgb, var(--cream) 20%, transparent)' }} />}
        </React.Fragment>
      ))}
    </div>
  );
}

function Builder({ onGenerate }) {
  const [code, setCode] = React.useState('R2L-JUN-A4F2');
  const [topic, setTopic] = React.useState(0);
  const step = !code ? 0 : 1;
  return (
    <div>
      <Stepper step={step} />
      <div style={{ marginBottom: 24 }}>
        <label className="fx-label">Bước 1 — Nhập mã học sinh <span style={{ color: 'var(--danger)' }}>*</span></label>
        <input className="fx-field" value={code} onChange={(e) => setCode(e.target.value.toUpperCase())} placeholder="VD: R2L-JUN-A4F2" style={{ fontSize: 'var(--text-lg)', textTransform: 'uppercase' }} />
      </div>
      <Card style={{ marginBottom: 24, background: 'color-mix(in srgb, var(--gold) 8%, transparent)', borderColor: 'var(--accent-border)', padding: 16 }}>
        <p style={{ fontSize: 'var(--text-sm)', fontWeight: 600, color: 'var(--gold-light)', margin: 0 }}>Mã học sinh là gì?</p>
        <p style={{ marginTop: 4, fontSize: 'var(--text-sm)', color: 'var(--cream-muted)', lineHeight: 1.6 }}>Mã riêng của bé — Felix gửi qua Zalo. Chưa có mã? Nhắn Zalo Felix.</p>
      </Card>
      <label className="fx-label">Bước 2 — Con thích chủ đề gì? <span style={{ color: 'var(--danger)' }}>*</span></label>
      <div style={{ display: 'grid', gridTemplateColumns: 'repeat(4, 1fr)', gap: 10, marginTop: 8 }} className="topic-grid">
        {TOPICS.map(([emoji, label], i) => (
          <TopicTile key={i} emoji={emoji} label={label} selected={topic === i} onClick={() => setTopic(i)} />
        ))}
      </div>
      <div style={{ marginTop: 28 }}>
        <Button size="lg" block disabled={!code} onClick={() => onGenerate(TOPICS[topic])}>Tạo bài học cho con</Button>
      </div>
    </div>
  );
}

function Generating() {
  const stages = [
    'Đang viết câu chuyện riêng cho con...',
    'Đang chuẩn bị phần nghe cho câu chuyện...',
    'Đang ghi âm các cụm câu để con tập đọc...',
    'Đang chuẩn bị nhiệm vụ web cho con...',
  ];
  const [s, setS] = React.useState(0);
  React.useEffect(() => {
    const t = setInterval(() => setS((x) => Math.min(x + 1, stages.length - 1)), 1400);
    return () => clearInterval(t);
  }, []);
  return (
    <Card variant="raised" style={{ textAlign: 'center', padding: 36 }}>
      <ProgressBar gradient value={30 + s * 20} />
      <p style={{ marginTop: 24, fontSize: 'var(--text-lg)', fontWeight: 600, color: 'var(--cream)' }}>{stages[s]}</p>
      <p style={{ marginTop: 12, fontSize: 'var(--text-sm)', color: 'var(--cream-dim)' }}>Lần đầu có thể chậm hơn một chút vì máy chủ cần khởi động.</p>
    </Card>
  );
}

function Result({ topic, onReset }) {
  return (
    <Card style={{ textAlign: 'center', padding: 36, background: 'color-mix(in srgb, var(--gold) 10%, transparent)', borderColor: 'var(--accent-border)' }}>
      <h2 style={{ fontFamily: 'var(--font-display)', fontWeight: 700, fontSize: 'var(--text-h3)', color: 'var(--cream)' }}>Bài học của bé Na đã sẵn sàng</h2>
      <div style={{ marginTop: 16, color: 'var(--cream-muted)' }}>
        <p>Chủ đề: <strong style={{ color: 'var(--cream)' }}>{topic[1]}</strong></p>
        <p>Tên truyện: <strong style={{ color: 'var(--cream)' }}>Chú Mèo Lạc Giữa Vì Sao</strong></p>
      </div>
      <div style={{ marginTop: 24 }}><Button size="lg">Mở bài học</Button></div>
      <div style={{ marginTop: 20 }}><Button variant="ghost" onClick={onReset}>Tạo bài mới</Button></div>
    </Card>
  );
}

function Leaderboard({ onBack }) {
  const rows = [
    ['🥇', 'Bảo Nam', 'legend', 1820], ['🥈', 'Minh Anh', 'diamond', 1640], ['🥉', 'Khánh Vy', 'gold', 1390],
    ['4', 'Gia Hân', 'gold', 1120], ['5', 'Bé Na', 'silver', 980], ['6', 'Tuấn Kiệt', 'bronze', 640],
  ];
  return (
    <div>
      <div style={{ display: 'flex', alignItems: 'center', gap: 12, marginBottom: 20 }}>
        <Button variant="ghost" onClick={onBack}>← Quay lại</Button>
        <h2 style={{ fontFamily: 'var(--font-display)', fontWeight: 700, fontSize: 'var(--text-h3)', color: 'var(--cream)', margin: 0 }}>Bảng xếp hạng</h2>
      </div>
      <div style={{ display: 'grid', gap: 10 }}>
        {rows.map(([rank, name, tier, xp], i) => (
          <Card key={i} style={{ display: 'flex', alignItems: 'center', gap: 14, padding: 14 }}>
            <span style={{ width: 28, textAlign: 'center', fontWeight: 700, color: 'var(--cream)' }}>{rank}</span>
            <span style={{ flex: 1, fontWeight: 600, color: 'var(--cream)' }}>{name}</span>
            <RankBadge tier={tier} />
            <span style={{ fontWeight: 700, color: 'var(--gold)', fontSize: 'var(--text-sm)', minWidth: 64, textAlign: 'right' }}>{xp} XP</span>
          </Card>
        ))}
      </div>
    </div>
  );
}

function App() {
  const [phase, setPhase] = React.useState('build'); // build | generating | result | leaderboard
  const [topic, setTopic] = React.useState(TOPICS[0]);
  const generate = (t) => { setTopic(t); setPhase('generating'); setTimeout(() => setPhase('result'), 5200); };
  return (
    <React.Fragment>
      <AppHeader onLeaderboard={() => setPhase('leaderboard')} />
      <main style={{ maxWidth: phase === 'leaderboard' ? '40rem' : '36rem', margin: '0 auto', padding: '40px var(--gutter) 80px', width: '100%', boxSizing: 'border-box' }}>
        {phase === 'build' && <Builder onGenerate={generate} />}
        {phase === 'generating' && <Generating />}
        {phase === 'result' && <Result topic={topic} onReset={() => setPhase('build')} />}
        {phase === 'leaderboard' && <Leaderboard onBack={() => setPhase('build')} />}
      </main>
    </React.Fragment>
  );
}

Object.assign(window, { R2LApp: App });
})();
